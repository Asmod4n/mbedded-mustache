option(MUSTACHE_REQUIRE_SIMDJSON "Fail the configure step when simdjson is missing" OFF)
set(MUSTACHE_SIMDJSON_SOURCE_DIR "" CACHE PATH
  "Source directory of simdjson, built when find_package(simdjson) finds none")
if(MUSTACHE_REQUIRE_SIMDJSON)
  find_package(simdjson REQUIRED)
else()
  find_package(simdjson QUIET)
endif()
if(NOT simdjson_FOUND AND MUSTACHE_SIMDJSON_SOURCE_DIR)
  set(SIMDJSON_CXX_STANDARD ${CMAKE_CXX_STANDARD} CACHE STRING "C++ standard of the simdjson build")
  add_subdirectory(${MUSTACHE_SIMDJSON_SOURCE_DIR} simdjson EXCLUDE_FROM_ALL SYSTEM)
  target_compile_options(simdjson PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/WX->)
  set(simdjson_FOUND TRUE)
endif()

set(MUSTACHE_PFR_SOURCE_DIR "" CACHE PATH
  "Source directory of Boost.PFR, used when find_package(Boost) finds none")
find_package(Boost CONFIG QUIET)
if(NOT Boost_FOUND AND MUSTACHE_PFR_SOURCE_DIR)
  add_library(boost_pfr INTERFACE)
  target_include_directories(boost_pfr SYSTEM INTERFACE ${MUSTACHE_PFR_SOURCE_DIR}/include)
  add_library(Boost::pfr ALIAS boost_pfr)
endif()
if(MUSTACHE_HAS_REFLECTION OR Boost_FOUND OR TARGET Boost::pfr)
  add_executable(typed typed.cpp)
  target_link_libraries(typed PRIVATE mustache::mustache GTest::gtest_main)
  if(Boost_FOUND)
    target_link_libraries(typed PRIVATE Boost::headers)
  elseif(TARGET Boost::pfr)
    target_link_libraries(typed PRIVATE Boost::pfr)
  endif()
  gtest_discover_tests(typed)
endif()
